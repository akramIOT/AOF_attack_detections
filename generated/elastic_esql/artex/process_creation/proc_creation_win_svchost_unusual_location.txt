# Svchost Running Outside the Windows System Folder
# Sigma id eae4b702-9723-5393-8d23-37295c6d40ea, pipeline ecs_windows
from * metadata _id, _index, _version | where ends_with(winlog.event_data.TargetProcessName, "\\svchost.exe") and not (starts_with(winlog.event_data.TargetProcessName, "C:\\Windows\\System32\\") or starts_with(winlog.event_data.TargetProcessName, "C:\\Windows\\SysWOW64\\") or starts_with(winlog.event_data.TargetProcessName, "C:\\Windows\\WinSxS\\"))
