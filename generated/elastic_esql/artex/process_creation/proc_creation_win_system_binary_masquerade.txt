# Windows System Binary Name Running From the Wrong Folder
# Sigma id d7ffbe3f-7838-53ca-89cc-d441a1088634, pipeline ecs_windows
from * metadata _id, _index, _version | where (ends_with(winlog.event_data.TargetProcessName, "\\smss.exe") or ends_with(winlog.event_data.TargetProcessName, "\\winlogon.exe") or ends_with(winlog.event_data.TargetProcessName, "\\dllhost.exe") or ends_with(winlog.event_data.TargetProcessName, "\\sihost.exe")) and not (starts_with(winlog.event_data.TargetProcessName, "C:\\Windows\\System32\\") or starts_with(winlog.event_data.TargetProcessName, "C:\\Windows\\SysWOW64\\")) or ends_with(winlog.event_data.TargetProcessName, "\\MicrosoftEdgeUpdate.exe") and not (winlog.event_data.TargetProcessName like "*\\Microsoft\\EdgeUpdate\\*" or winlog.event_data.TargetProcessName like "*\\Microsoft\\Temp\\*")
