# Remote tasklist Fan-Out From One Host
# Sigma id c53722c6-23d3-5460-a002-2d1b034e2809, pipeline ecs_windows
from * metadata _id, _index, _version | where (ends_with(winlog.event_data.TargetProcessName, "\\tasklist.exe") or winlog.event_data.TargetProcessFileOriginalName=="tasklist.exe" or winlog.event_data.TargetProcessFilename=="tasklist.exe") and (winlog.event_data.TargetProcessCommandLine like "* -s *" or winlog.event_data.TargetProcessCommandLine like "* /s *" or winlog.event_data.TargetProcessCommandLine like "* –s *" or winlog.event_data.TargetProcessCommandLine like "* —s *" or winlog.event_data.TargetProcessCommandLine like "* ―s *")
| eval timebucket=date_trunc(10minutes, @timestamp) | stats event_count=count() by timebucket, winlog.event_data.Computer
| where event_count >= 10
