# CloudWatch Logs Searched for Secrets
# Sigma id 510a5a9c-a361-513d-b3f4-52af66cdf912, pipeline ecs_windows
from * metadata _id, _index, _version | where eventSource=="logs.amazonaws.com" and eventName=="FilterLogEvents" and (requestParameters.filterPattern like "*password*" or requestParameters.filterPattern like "*AKIA*" or requestParameters.filterPattern like "*secret*" or requestParameters.filterPattern like "*token*")
